# --- Security ---
pyjwt~=2.9.0
passlib[bcrypt]~=1.7.4
# Pinned <4.1: passlib 1.7.4 (last released 2020) reads bcrypt's
# `__about__.__version__` to detect the backend; bcrypt>=4.1 removed
# that module, which passlib silently traps as a warning and then
# mis-negotiates the backend, breaking every password hash with
# "ValueError: password cannot be longer than 72 bytes" — even for
# short passwords. A fresh `pip install` with no upper bound resolves
# straight to the latest bcrypt and hits this. Caught by actually
# running tests/unit/test_security.py against a clean install rather
# than trusting requirements.txt's ranges alone.
bcrypt<4.1.0
